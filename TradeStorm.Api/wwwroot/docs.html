SignalR ChatHub (Двосторонній WebSocket чат)

LoopMarket використовує SignalR хаб (/chathub) для миттєвої доставки повідомлень, передачі статусів набору тексту («друкує...»), відстеження присутності в мережі та спливаючих сповіщень про нові угоди Loop Захист.

JavaScript / TypeScript Client 
// npm i @microsoft/signalr
import * as signalR from "@microsoft/signalr";

const token = localStorage.getItem("token");

const connection = new signalR.HubConnectionBuilder()
    .withUrl("/chathub", {
        accessTokenFactory: () => token
    })
    .withAutomaticReconnect([0, 2000, 5000, 10000])
    .build();

// Слухаємо нові вхідні повідомлення
connection.on("ReceiveMessage", (senderId, receiverId, text) => {
    console.log(`💬 Повідомлення від ${senderId}: ${text}`);
    renderMessageInChat(senderId, text);
});

// Слухаємо набір тексту
connection.on("UserIsTyping", (senderId) => {
    showTypingIndicator(senderId);
});

// Підключення
async function start() {
    await connection.start();
    await connection.invoke("RegisterUser", 1);
    console.log("✅ SignalR ChatHub підключено!");
}
start();

// Відправка повідомлення
async function sendChatMessage(partnerId, text) {
    await connection.invoke("SendMessage", "1", partnerId.toString(), text);
}
C# (.NET 10 Client) 
using Microsoft.AspNetCore.SignalR.Client;

var hubConnection = new HubConnectionBuilder()
    .WithUrl("http://localhost:65083/chathub", options =>
    {
        options.AccessTokenProvider = () => Task.FromResult(jwtToken);
    })
    .WithAutomaticReconnect()
    .Build();

// Обробник отримання повідомлення
hubConnection.On<string, string, string>("ReceiveMessage", (senderId, receiverId, text) =>
{
    Console.WriteLine($"[Чат] {senderId} -> {receiverId}: {text}");
});

await hubConnection.StartAsync();
await hubConnection.InvokeAsync("RegisterUser", 1);

// Відправка
await hubConnection.InvokeAsync("SendMessage", "1", "2", "Вітаю!");

Специфікація методів ChatHub

Напрямок Метод / Подія Параметри Опис
Клієнт -> Сервер RegisterUser(userId) int userId Прив'язує SignalR ConnectionId до ID користувача в пам'яті сервера.
Клієнт -> Сервер SendMessage(senderId, receiverId, text) string, string, string Зберігає повідомлення в БД, відправляє в Telegram та викликає ReceiveMessage.

Сервер -> Клієнт ReceiveNotification NotificationDto notification Миттєве сповіщення про нове замовлення або зміну статусу безпечної угоди.

Архітектура та Життєвий Цикл «Loop Захист» (Escrow)

«Loop Захист» — сервіс безпечних угод для товарів та послуг, що гарантує фінансову безпеку обом сторонам. Кошти заморожуються на платформі й перераховуються виконавцю лише після перевірки та підтвердження завершення робіт клієнтом.

1
Оформлення замовлення (POST /api/orders/create-service)
Замовник заповнює параметри послуги: точну дату, час, адресу виконання та технічний опис. Замовленню присвоюється статус Confirmed (кошти заморожено на рахунку Loop Захист).
2
Миттєве сповіщення виконавця
Майстер або продавець отримує сповіщення через SignalR у реальному часі та дублююче push-повідомлення в офіційний Telegram-бот. Замовлення з'являється у вкладці «Мої продажі / замовлення».
3
Виконання робіт або доставка (Статус: Shipped / InDelivery)
При відправці товару продавець додає номер ТТН Нової Пошти через PATCH /api/orders/{id}/status. Для послуг майстер прибуває за адресою у визначений час та виконує заявлений обсяг робіт.
4
Прийом робіт замовником та виплата коштів (Статус: Completed)
Замовник перевіряє якість та тисне «Підтвердити виконання». Платформа автоматично виплачує кошти на картку/баланс виконавця за вирахуванням сервісної комісії 5%.
⚖️
Арбітраж та Спірні ситуації (Статус: Disputed)
Якщо виконавець не з'явився, або товар виявився неналежної якості, замовник може відкрити диспут. Кошти залишаються заблокованими до винесення рішення модератором або повертаються замовнику в 100% обсязі.

Коди Відповідей HTTP & Обробка Помилок

Всі відповіді сервера повертаються у стандартному форматі JSON. При помилках тіло містить поле error або message з чітким описом причини.

// 2. Фоновий поллінг: сайт сам авторизує користувача після натискання в боті
function startVerificationPolling() {
    if (pollInterval) return;
    pollInterval = setInterval(async () => {
        const email = document.getElementById('regEmail')?.value;
        const password = document.getElementById('regPassword')?.value;
        if (!email || !password) return;

        const res = await fetch('/api/users/login', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, password })
        });
        if (res.ok) {
            const data = await res.json();
            if (data.isPhoneVerified) {
                clearInterval(pollInterval);
                localStorage.setItem('loopToken', data.token);
                window.location.href = '/index.html';
            }
        }
    }, 2500);
}

async function reverseGeocodeLocation(lat, lng) {
    const res = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lng}&accept-language=uk`);
    if (res.ok) {
        const data = await res.json();
        const road = data.address?.road || '';
        const house = data.address?.house_number || '';
        const city = data.address?.city || data.address?.town || 'Кременчук';
        const formatted = [road, house].filter(Boolean).join(' ') + (road ? `, ${city}` : city);
        document.getElementById('location').value = formatted;
    }
}

public async Task<(bool Success, string? Error, Listing? Listing)> UpdateListingAsync(
    int listingId, int userId, UpdateListingDto dto, bool isAdmin = false)
{
    var listing = await _listingRepository.GetByIdAsync(listingId);
    if (listing == null) return (false, "Оголошення не знайдено", null);

    // Дозволено власнику або Оператору/Адміністратору
    if (!isAdmin && listing.SellerId != userId)
        throw new UnauthorizedAccessException("Ви не маєте права редагувати чуже оголошення.");

    listing.Title = dto.Title.Trim();
    listing.Description = dto.Description.Trim();
    listing.Price = dto.Price;
    listing.CategoryId = dto.CategoryId;
    listing.Condition = dto.Condition;
    listing.Location = string.IsNullOrWhiteSpace(dto.Location) ? "Кременчук" : dto.Location.Trim();
    listing.IsShippingAvailable = dto.IsShippingAvailable;

    // Збереження існуючих фото або встановлення фірмової заглушки
    if (dto.ImageUrls != null)
    {
        listing.ImageUrl = dto.ImageUrls.Any() 
            ? string.Join(";", dto.ImageUrls) 
            : (dto.CategoryId == 8 ? "/images/placeholder-service.svg" : "/images/placeholder.svg");
    }

    await _listingRepository.UpdateAsync(listing);
    return (true, null, listing);
}

[Authorize]
public class ChatHub : Hub
{
    public async Task SendMessage(int receiverId, int? listingId, string text)
    {
        var senderId = int.Parse(Context.UserIdentifier!);

        // 1. Збереження повідомлення у БД
        var msg = new Message { SenderId = senderId, ReceiverId = receiverId, ListingId = listingId, Text = text, CreatedAt = DateTime.UtcNow };
        _context.Messages.Add(msg);
        await _context.SaveChangesAsync();

        // 2. Миттєва доставка через WebSocket у браузер отримувача
        await Clients.User(receiverId.ToString()).SendAsync("ReceiveMessage", new {
            id = msg.Id, senderId = senderId, text = text, createdAt = msg.CreatedAt
        });

        // 3. Якщо користувач офлайн — дублюємо в Telegram
        var receiver = await _context.Users.FindAsync(receiverId);
        if (receiver != null && receiver.TelegramChatId != 0 && receiver.ReceiveTelegramNotifications)

                receiver.TelegramChatId, 
                $"💬 Нове повідомлення на LoopMarket!\n{text}\n\nВідповісти в чаті"
            );
        }
    }
}

[Authorize(Roles = "Admin")]
[HttpPost("users/{userId}/ban")]
public async Task<IActionResult> BanUser(int userId, [FromBody] BanUserDto dto)
{
    var user = await _context.Users.FindAsync(userId);
    if (user == null) return NotFound(new { error = "Користувача не знайдено" });

    // 1. Блокування користувача
    user.IsBanned = true;
    user.BanReason = dto.Reason;
    user.RefreshToken = null; // Миттєве анулювання активних сесій

    // 2. Запис у незмінний журнал аудиту
    var adminId = int.Parse(User.FindFirstValue(ClaimTypes.NameIdentifier)!);
    _context.AdminAuditLogs.Add(new AdminAuditLog {
        AdminUserId = adminId,
        Action = "BanUser",
        Target = $"User #{userId} ({user.Email})",
        Details = $"Причина: {dto.Reason}",
        IpAddress = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "Unknown",
        CreatedAt = DateTime.UtcNow
    });

    await _context.SaveChangesAsync();
    return Ok(new { message = $"Користувача #{userId} успішно заблоковано" });
}

function toggleTheme() {
