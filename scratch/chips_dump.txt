            const wrap = document.getElementById('quickChips');

                    '📅 Коли у вас є вільний час для виконання?',
                    '📍 Чи можливий виїзд за моєю адресою?',
                    '🤝 Яка остання ціна? Можливий торг?'
                ];
            } else {
                chips = [
                    '📦 Вітаю! Оголошення ще актуальне?',

            wrap.innerHTML = chips.map(c => `
